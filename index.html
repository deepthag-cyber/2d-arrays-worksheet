Declaring a 2D Array

A 2D array organises data in rows and columns — like a table. Every cell is accessed using two indices separated by a comma: Scores[I, J].

Rows are the outer index (I); columns are the inner index (J). Index counting starts at 0.
Pseudocode
DECLARE Scores : ARRAY[0:2, 0:3] OF INTEGER

// Scores[0, 0]  Scores[0, 1]  Scores[0, 2]  Scores[0, 3]
// Scores[1, 0]  Scores[1, 1]  Scores[1, 2]  Scores[1, 3]
// Scores[2, 0]  Scores[2, 1]  Scores[2, 2]  Scores[2, 3]
Python
# Create a 3×4 2D list (zeros)
Scores = [[0] * 4 for _ in range(3)]

# Access: Scores[row][col]
# e.g. Scores[1][2] → row 1, col 2

Initialising a 2D Array

Initialisation assigns values to array cells — either by hard-coding values directly, or using a nested FOR loop.

Pseudocode — hard-coded
DECLARE Grid : ARRAY[0:1, 0:2] OF INTEGER
Grid[0, 0] ← 10   Grid[0, 1] ← 20   Grid[0, 2] ← 30
Grid[1, 0] ← 40   Grid[1, 1] ← 50   Grid[1, 2] ← 60

Pseudocode — nested FOR loop (fills 1 to 9)
DECLARE Grid : ARRAY[0:2, 0:2] OF INTEGER
DECLARE I, J : INTEGER

FOR I ← 0 TO 2
    FOR J ← 0 TO 2
        Grid[I, J] ← I * 3 + J + 1
    NEXT J
NEXT I
Python
Grid = [[0] * 3 for _ in range(3)]
for I in range(3):
    for J in range(3):
        Grid[I][J] = I * 3 + J + 1
print(Grid) # [[1,2,3],[4,5,6],[7,8,9]]

Populating a 2D Array with User Input

The nested loop asks the user for each cell one at a time. The outer loop picks the row; the inner loop picks the column.

Pseudocode
DECLARE Marks : ARRAY[0:2, 0:3] OF INTEGER
DECLARE I, J : INTEGER

FOR I ← 0 TO 2
    FOR J ← 0 TO 3
        OUTPUT "Enter mark for row ", I,
               " col ", J
        INPUT Marks[I, J]
    NEXT J
NEXT I
Python
ROWS, COLS = 3, 4
Marks = [[0] * COLS for _ in range(ROWS)]

for I in range(ROWS):
    for J in range(COLS):
        Marks[I][J] = int(input(
            f"Enter mark for row {I} col {J}: "))
for Row in Marks:
    print(Row)

Bubble Sort — Each Row (REPEAT…UNTIL)

This version uses a REPEAT…UNTIL loop with a Swap flag — it stops early if no swaps occurred in a pass, making it more efficient.

Pseudocode — bubble sort each row
DECLARE Data : ARRAY[0:2, 0:3] OF INTEGER
DECLARE I, Index, Last : INTEGER
DECLARE Temp : INTEGER
DECLARE Swap : BOOLEAN

// Assume Data[][] is already populated

FOR I ← 0 TO 2          // for each row
    Last ← 3             // last index to compare
    REPEAT
        Swap ← FALSE
        FOR Index ← 0 TO Last - 1
            IF Data[I, Index] > Data[I, Index + 1]
            THEN
                Temp              ← Data[I, Index]
                Data[I, Index]    ← Data[I, Index + 1]
                Data[I, Index + 1] ← Temp
                Swap ← TRUE
            ENDIF
        NEXT Index
        Last ← Last - 1
    UNTIL (NOT Swap) OR Last = 0
NEXT I
Python
Data = [[4, 1, 3, 2],
        [9, 5, 7, 6],
        [2, 8, 1, 4]]
ROWS, COLS = 3, 4

for I in range(ROWS):
    Last = COLS - 1
    while True:
        Swap = False
        for Index in range(Last):
            if Data[I][Index] > Data[I][Index+1]:
                Data[I][Index], Data[I][Index+1] = \
                    Data[I][Index+1], Data[I][Index]
                Swap = True
        Last -= 1
        if (not Swap) or Last == 0:
            break

for Row in Data:
    print(Row)

Linear Search Across All Rows

Scans every element row by row. A Found flag tracks whether the target was located. All occurrences are reported.

Pseudocode
DECLARE Data : ARRAY[0:2, 0:2] OF INTEGER
DECLARE I, J, Target : INTEGER
DECLARE Found : BOOLEAN

OUTPUT "Enter value to search:"
INPUT Target
Found ← FALSE

FOR I ← 0 TO 2
    FOR J ← 0 TO 2
        IF Data[I, J] = Target THEN
            OUTPUT "Found at row ", I,
                   " col ", J
            Found ← TRUE
        ENDIF
    NEXT J
NEXT I

IF Found = FALSE THEN
    OUTPUT "Value not found"
ENDIF
Python
Data = [[5, 8, 3],
        [7, 2, 6],
        [4, 9, 3]]

Target = int(input('Enter value to search: '))
Found = False

for I in range(len(Data)):
    for J in range(len(Data[I])):
        if Data[I][J] == Target:
            print(f"Found at row {I} col {J}")
            Found = True

if not Found:
    print("Value not found")
